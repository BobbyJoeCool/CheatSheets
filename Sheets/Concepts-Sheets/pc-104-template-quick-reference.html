Binary Search (First True)

(lo, hi) ← (low, high)
while lo < hi
    mid ← lo + (hi - lo) div 2
    if ok(mid)
        hi ← mid
    else
        lo ← mid + 1
return lo

Sliding Window (Longest)

l ← 0
for r ← 0 to n - 1
    add(a[r])
    while invalid()
        remove(a[l])
        l ← l + 1
    best ← max(best, r - l + 1)

DP Memo

memo ← new Map()
function f(s)
    if isBase(s) return base(s)
    if memo.has(s) return memo[s]
    memo[s] ← best of f(next)
              over each choice
    return memo[s]

BFS (Shortest Steps)

queue ← new Queue([start])
seen ← new Set([start])
steps ← 0
while not queue.isEmpty()
    repeat length(queue) times
        u ← queue.dequeue()
        if u = goal return steps
        for each v in neighbors(u)
            if not seen.has(v)
                seen.add(v)
                queue.enqueue(v)
    steps ← steps + 1
return -1

Dijkstra

dist ← new Array(n, ∞)
dist[s] ← 0
heap ← new MinHeap([(0, s)])
while not heap.isEmpty()
    (d, u) ← heap.pop()
    if d > dist[u] continue
    for each (v, w) in adj[u]
        if d + w < dist[v]
            dist[v] ← d + w
            heap.push((dist[v], v))
return dist

Backtracking

function bt(start, path)
    if done(path)
        out.append(copy(path))
        return
    for i ← start to n - 1
        if not valid(i) continue
        path.append(a[i])
        bt(i + 1, path)
        path.pop()

Union-Find

function find(x)
    while parent[x] ≠ x
        parent[x] ← parent[parent[x]]
        x ← parent[x]
    return x
function union(a, b)
    (ra, rb) ← (find(a), find(b))
    if ra = rb return false
    if size[ra] < size[rb]
        (ra, rb) ← (rb, ra)
    parent[rb] ← ra
    size[ra] ← size[ra] + size[rb]
    return true

DFS (Components)

function dfs(u)
    seen.add(u)
    for each v in adj[u]
        if not seen.has(v)
            dfs(v)
count ← 0
for each u in nodes
    if not seen.has(u)
        dfs(u)
        count ← count + 1

⚠ Common Mistakes

  • Binary search: hi ← mid needs lo < hi
  • BFS: mark seen when enqueuing

✓ Tips & Best Practices

  • Windows: record longest after shrinking
  • Backtracking: copy the path
