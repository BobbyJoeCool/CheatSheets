Registering & Configuring Tasks

val gen = tasks.register<Copy>("copyDocs") {
    from("docs"); into(layout.buildDirectory.dir("site"))
}
tasks.named("assemble") { dependsOn(gen) }
tasks.withType<JavaCompile>().configureEach {
    options.encoding = "UTF-8"
}
tasks.register("hello") {
    val v = project.version.toString()
    doLast { println("v$v") }
}

Custom Task Class

abstract class Stamp : DefaultTask() {
    @get:Input abstract val text: Property<String>
    @get:OutputFile abstract val out: RegularFileProperty
    @TaskAction fun run() =
        out.get().asFile.writeText(text.get())
}
tasks.register<Stamp>("stamp") {
    text = "built"
    out = layout.buildDirectory.file("stamp.txt")
}
